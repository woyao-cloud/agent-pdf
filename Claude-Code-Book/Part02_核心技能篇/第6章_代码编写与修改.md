# 之前def process_items(items, filter_func=None):
    result = []    for item in items:        if filter_func is None or filter_func(item):
            result.append(item)    return result# 之后from typing import TypeVar, Callable, List, Optional
T = TypeVar('T')def process_items(
    items: List[T],    filter_func: Optional[Callable[[T], bool]] = None
) -> List[T]:    """处理列表项，可选地过滤。        Args:        items: 输入列表        filter_func: 过滤函数，如果为 None 则不过滤        Returns:        过滤后的列表    """    result = []    for item in items:        if filter_func is None or filter_func(item):
            result.append(item)    return result
