**迁移学习的关键步骤**：

1. 冻结预训练模型（trainable = False）
2. 先只训练新增的头部（分类器）
3. 然后解冻（unfreeze）部分层进行微调
4. 用极低的学习率（通常 1e-5 ~ 1e-4）

### 1.3 图像分割

```python
# U-Net 架构示例
inputs = tf.keras.layers.Input(shape=(256, 256, 3))

# Encoder
c1 = tf.keras.layers.Conv2D(64, 3, activation='relu', padding='same')(inputs)
c1 = tf.keras.layers.Conv2D(64, 3, activation='relu', padding='same')(c1)
p1 = tf.keras.layers.MaxPooling2D(2)(c1)

# Decoder
u6 = tf.keras.layers.Conv2DTranspose(64, 2, strides=2)(p1)
u6 = tf.keras.layers.Concatenate()([u6, c1])
c6 = tf.keras.layers.Conv2D(64, 3, activation='relu', padding='same')(u6)

outputs = tf.keras.layers.Conv2D(1, 1, activation='sigmoid')(c6)
model = tf.keras.Model(inputs, outputs)
```

> ⚠️ **陷阱**：图像分割的 mask 是二值还是多类？多类用 Softmax，二值用 Sigmoid。损失函数也不同：二值用 BinaryCrossentropy，多类用 SparseCategoricalCrossentropy。

### 1.4 常见 CV 预处理

```python
def preprocess_image(image, label):
    # 归一化到 [0, 1]
    image = tf.image.convert_image_dtype(image, tf.float32)
    # 随机增强
    image = tf.image.random_flip_left_right(image)
    image = tf.image.random_brightness(image, 0.1)
    image = tf.image.random_contrast(image, 0.9, 1.1)
    # 标准化（ImageNet 统计）
    mean = tf.constant([0.485, 0.456, 0.406])
    std = tf.constant([0.229, 0.224, 0.225])
    image = (image - mean) / std
    return image, label
```

### 2.2 序列到序列（Seq2Seq）

```python
# 简单 seq2seq（用于理解原理）
class Seq2Seq(tf.keras.Model):
    def __init__(self, vocab_size, embed_dim, units):
        super().__init__()
        self.embedding = tf.keras.layers.Embedding(vocab_size, embed_dim)
        self.encoder = tf.keras.layers.GRU(units, return_state=True)
        self.decoder = tf.keras.layers.GRU(units, return_sequences=True)
        self.dense = tf.keras.layers.Dense(vocab_size)
    
    def call(self, encoder_input, decoder_input, training=False):
        _, state = self.encoder(self.embedding(encoder_input))
        decoder_output = self.decoder(self.embedding(decoder_input), initial_state=state)
        return self.dense(decoder_output)
```

### 2.3 Transformer 实战

```python
# 简化版 Transformer Encoder
class TransformerBlock(tf.keras.layers.Layer):
    def __init__(self, embed_dim, num_heads, ff_dim, rate=0.1):
        super().__init__()
        self.att = tf.keras.layers.MultiHeadAttention(num_heads, embed_dim)
        self.ffn = tf.keras.Sequential([
            tf.keras.layers.Dense(ff_dim, activation='relu'),
            tf.keras.layers.Dense(embed_dim)
        ])
        self.layernorm1 = tf.keras.layers.LayerNormalization(epsilon=1e-6)
        self.layernorm2 = tf.keras.layers.LayerNormalization(epsilon=1e-6)
        self.dropout1 = tf.keras.layers.Dropout(rate)
        self.dropout2 = tf.keras.layers.Dropout(rate)
    
    def call(self, x, training=False):
        attn_output = self.att(x, x)
        x = self.layernorm1(x + self.dropout1(attn_output, training=training))
        ffn_output = self.ffn(x)
        return self.layernorm2(x + self.dropout2(ffn_output, training=training))
```

> ✅ **最佳实践**：NLP 场景优先用 Hugging Face Transformers（虽然后端是 PyTorch，但 TF 版本也能用）。自己写 Transformer 主要是为了学习原理。

## 3. 推荐系统

### 3.1 经典双塔模型

```python
# 用户塔
user_input = tf.keras.layers.Input(shape=(1,))
user_embed = tf.keras.layers.Embedding(100000, 32)(user_input)
user_embed = tf.keras.layers.Flatten()(user_embed)
user_embed = tf.keras.layers.Dense(64, activation='relu')(user_embed)

# 物品塔
item_input = tf.keras.layers.Input(shape=(1,))
item_embed = tf.keras.layers.Embedding(100000, 32)(item_input)
item_embed = tf.keras.layers.Flatten()(item_embed)
item_embed = tf.keras.layers.Dense(64, activation='relu')(item_embed)

# 交互
dot = tf.keras.layers.Dot(axes=1)([user_embed, item_embed])
outputs = tf.keras.layers.Dense(1, activation='sigmoid')(dot)

model = tf.keras.Model([user_input, item_input], outputs)
model.compile(optimizer='adam', loss='binary_crossentropy', metrics=['auc'])
```

### 3.2 Deep & Wide

```python
# Wide 侧（记忆）
wide_input = tf.keras.layers.Input(shape=(10,))

# Deep 侧（泛化）
deep_input = tf.keras.layers.Input(shape=(10,))
deep = tf.keras.layers.Dense(64, activation='relu')(deep_input)
deep = tf.keras.layers.Dense(32, activation='relu')(deep)

# 合并
merged = tf.keras.layers.Concatenate()([wide_input, deep])
output = tf.keras.layers.Dense(1, activation='sigmoid')(merged)

model = tf.keras.Model([wide_input, deep_input], output)
```

def create_sequences(data, seq_length):
    xs, ys = [], []
    for i in range(len(data) - seq_length):
        xs.append(data[i:i+seq_length])
        ys.append(data[i+seq_length])
    return np.array(xs), np.array(ys)

# 添加时间特征
def add_time_features(df):
    df['hour_sin'] = np.sin(2 * np.pi * df['hour'] / 24)
    df['hour_cos'] = np.cos(2 * np.pi * df['hour'] / 24)
    df['day_sin'] = np.sin(2 * np.pi * df['day'] / 7)
    return df
