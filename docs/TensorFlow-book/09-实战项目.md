## 项目 1: MNIST → TFLite 端侧部署

### 1.1 项目概述

从零训练一个手写数字识别模型，导出为 TFLite 并部署到 Android/iOS。项目虽小，但覆盖完整流程：数据→模型→训练→优化→转换→部署。

### 1.2 数据加载与探索

```python
import tensorflow as tf
import numpy as np
import matplotlib.pyplot as plt

# 加载 MNIST
(x_train, y_train), (x_test, y_test) = tf.keras.datasets.mnist.load_data()

print(f"训练集: {x_train.shape}, 标签: {y_train.shape}")
print(f"测试集: {x_test.shape}, 标签: {y_test.shape}")
print(f"像素范围: [{x_train.min()}, {x_train.max()}]")

# 可视化几个样本
fig, axes = plt.subplots(1, 5, figsize=(15, 3))
for i, ax in enumerate(axes):
    ax.imshow(x_train[i], cmap='gray')
    ax.set_title(f'Label: {y_train[i]}')
    ax.axis('off')
plt.savefig('samples.png')
```

### 1.3 数据预处理

```python
# 归一化 + 维度调整
def preprocess(x, y):
    x = tf.cast(x, tf.float32) / 255.0  # 归一化到 [0, 1]
    x = tf.expand_dims(x, axis=-1)       # 添加通道维度 (28, 28) -> (28, 28, 1)
    return x, y

train_ds = (tf.data.Dataset.from_tensor_slices((x_train, y_train))
    .shuffle(10000)
    .map(preprocess, num_parallel_calls=tf.data.AUTOTUNE)
    .batch(128)
    .prefetch(tf.data.AUTOTUNE))

test_ds = (tf.data.Dataset.from_tensor_slices((x_test, y_test))
    .map(preprocess)
    .batch(128))
```

### 1.4 构建模型

```python
def create_model():
    model = tf.keras.Sequential([
        # 输入: (28, 28, 1)
        tf.keras.layers.Conv2D(32, 3, activation='relu', input_shape=(28, 28, 1)),
        tf.keras.layers.MaxPooling2D(),
        tf.keras.layers.Conv2D(64, 3, activation='relu'),
        tf.keras.layers.MaxPooling2D(),
        tf.keras.layers.Conv2D(64, 3, activation='relu'),
        tf.keras.layers.Flatten(),
        tf.keras.layers.Dense(64, activation='relu'),
        tf.keras.layers.Dropout(0.25),
        tf.keras.layers.Dense(10)  # 输出 10 类
    ])
    return model

model = create_model()
model.summary()
```

### 1.5 训练模型

```python
model.compile(
    optimizer='adam',
    loss=tf.keras.losses.SparseCategoricalCrossentropy(from_logits=True),
    metrics=['accuracy']
)

history = model.fit(
    train_ds,
    epochs=10,
    validation_data=test_ds,
    callbacks=[
        tf.keras.callbacks.EarlyStopping(patience=3, restore_best_weights=True),
        tf.keras.callbacks.ModelCheckpoint('best_model.h5', save_best_only=True)
    ]
)
```

### 1.6 评估模型

```python
test_loss, test_acc = model.evaluate(test_ds)
print(f'测试集准确率: {test_acc:.4f}')

# 保存为 SavedModel
model.save('mnist_model')

# 导出为 TFLite
converter = tf.lite.TFLiteConverter.from_keras_model(model)
converter.optimizations = [tf.lite.Optimize.DEFAULT]
tflite_model = converter.convert()

with open('mnist.tflite', 'wb') as f:
    f.write(tflite_model)
```

### 1.7 验证 TFLite 模型

```python
# 加载并测试 TFLite 模型
interpreter = tf.lite.Interpreter(model_path='mnist.tflite')
interpreter.allocate_tensors()

input_index = interpreter.get_input_details()[0]['index']

# 测试一个样本

interpreter.set_tensor(input_index, test_image)
```

### 2.4 模型构建

```python
# 使用 TF Hub 的 BERT
bert_model_name = 'small_bert/bert_en_uncased_L-4_H-512_A-8'

bert_preprocess = hub.load('https://tfhub.dev/tensorflow/bert_en_uncased_preprocess/3')
bert_encoder = hub.load('https://tfhub.dev/tensorflow/bert_en_uncased_L-4_H-512_A-8/1')

# 构建分类模型
def build_classifier_model():
    text_input = tf.keras.layers.Input(shape=(), dtype=tf.string, name='text')
    preprocessed = bert_preprocess(text_input)
    encoder_outputs = bert_encoder(preprocessed)
    pooled = encoder_outputs['pooled_output']
    drop = tf.keras.layers.Dropout(0.1)(pooled)
    output = tf.keras.layers.Dense(1, activation='sigmoid', name='output')(drop)
    return tf.keras.Model(text_input, output)

model = build_classifier_model()
model.compile(optimizer='adam', loss='binary_crossentropy', metrics=['accuracy'])
```

### 3.3 数据加载（使用 tf.data）

```python
# 使用 TensorFlow Datasets
import tensorflow_datasets as tfds

def preprocess_image(image, label):
    image = tf.cast(image, tf.float32) / 255.0
    # 简化的预处理
    return image, label

# 加载数据
(datasets, info) = tfds.load('cifar10', with_info=True, as_supervised=True)
train_ds = datasets['train'].map(preprocess_image).shuffle(10000).batch(64)
test_ds = datasets['test'].map(preprocess_image).batch(64)

# 数据集分片（多 worker）
train_ds = train_ds.shard(num_shards=tf.distribute.get_strategy().num_replicas_in_sync, 
                          index=0)
```

### 3.4 构建模型

```python
# 构建 ResNet（简化版）
def resnet_block(x, filters, kernel_size=3, stride=1):
    y = tf.keras.layers.Conv2D(filters, kernel_size, stride=stride, padding='same')(x)
    y = tf.keras.layers.BatchNormalization()(y)
    y = tf.keras.layers.ReLU()(y)
    y = tf.keras.layers.Conv2D(filters, kernel_size, padding='same')(y)
    y = tf.keras.layers.BatchNormalization()(y)
    
    if stride != 1:
        x = tf.keras.layers.Conv2D(filters, 1, stride)(x)
    
    return tf.keras.layers.ReLU()(x + y)

def build_resnet():
    inputs = tf.keras.layers.Input(shape=(32, 32, 3))
    x = tf.keras.layers.Conv2D(64, 3, padding='same')(x)
    x = resnet_block(x, 64)
    x = resnet_block(x, 64, stride=2)  # 下采样
    x = resnet_block(x, 128)
    x = tf.keras.layers.GlobalAveragePooling2D()(x)
    outputs = tf.keras.layers.Dense(10, activation='softmax')(x)
    return tf.keras.Model(inputs, outputs)
```

### 3.5 分布式训练配置

```python
# 开启混合精度
from tensorflow.keras import mixed_precision
mixed_precision.set_global_policy('mixed_float16')

# 分布式策略
strategy = tf.distribute.MirroredStrategy()
print(f'使用 {strategy.num_replicas_in_sync} 个 GPU')

# 在策略作用域内构建模型
with strategy.scope():
    model = build_resnet()
    model.compile(
        optimizer=tf.keras.optimizers.Adam(1e-3),
        loss='sparse_categorical_crossentropy',
        metrics=['accuracy']
    )
```
