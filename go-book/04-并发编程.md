# 第 4 章 并发编程
> **本章导读**：Go 最强大的特性——goroutine 和 channel。Go 的并发模型基于 CSP（Communicating Sequential Processes），让你能轻松写出高并发代码。## 1. Goroutine
### 1.1 什么是 Goroutine
Goroutine 是 Go 的轻量级线程，由 Go 运行时管理：```go
func say(s string) {    for i := 0; i < 3; i++ {        time.Sleep(100 * time. Millisecond)        fmt.Println(s)    }}func main() {    go say("hello")  // 启动 goroutine    go say("world")  // 启动另一个    time.Sleep(1 * time.Second)  // 等待}```### 1.2 Goroutine vs 线程| 维度 | 线程 | Goroutine | |------|-----|----------| | 创建成本 | ~1MB 栈 | ~2KB 栈 | | 切换成本 | 内核态 | 用户态 | | 调度 | OS 抢占 | Go 协作 | | 数量限制 | 几千 | 十万级 |### 1.3 启动多个 Goroutine
```go
func worker(id int) {    fmt.Printf("Worker %d starting\n", id)    time.Sleep(time.Second)    fmt.Printf("Worker %d done\n", id)}func main() {    for i := 1; i <= 3; i++ {        go worker(i)    }    time.Sleep(2 * time.Second)}```## 2. Channel
### 2.1 创建和使用
```go
ch := make(chan int)  // 无缓冲channelch <- 10              // 发送v := <-ch              // 接收close(ch)              // 关闭// 带缓冲的channel
ch := make(chan int, 10)  // 缓冲区大小 10
```### 2.2 单向 Channel
```go
// send-onlyfunc send(ch chan<- int) {    ch <- 10}// receive-onlyfunc receive(ch <-chan int) {    v := <-ch    fmt.Println(v)}func main() {    ch := make(chan int)    go send(ch)    receive(ch)}```### 2.3 Select
同时等待多个 channel：```go
func main() {    ch1 := make(chan string)    ch2 := make(chan string)    go func() { ch1 <- "one" }()    go func() { ch2 <- "two" }()        for i := 0; i < 2; i++ {        select {        case msg1 := <-ch1:            fmt.Println(msg1)        case msg2 := <-ch2:            fmt.Println(msg2)        }    }}```> 💡 **进阶**：select + default 实现非阻塞。```go
select {case v := <-ch:    fmt.Println(v)default:    fmt.Println("no value ready")}```## 3. 同步原语
### 3.1 WaitGroup
等待一组 goroutine 完成：```go
import "sync"func worker(id int, wg *sync.WaitGroup) {    defer wg.Done()  // 完成时通知    fmt.Printf("Worker %d starting\n", id)    time.Sleep(time.Second)    fmt.Printf("Worker %d done\n", id)}func main() {    var wg sync.WaitGroup    for i := 1; i <= 5; i++ {        wg.Add(1)        go worker(i, &wg)    }    wg.Wait()  // 等待所有 worker 完成    fmt.Println("All workers done")}```### 3.2 Mutex
保护共享资源：```go
type SafeCounter struct {    mu    sync.Mutex    count int}func (c *SafeCounter) Inc() {    c.mu.Lock()    defer c.mu.Unlock()    c.count++}func (c *SafeCounter) Value() int {    c.mu.Lock()    defer c.mu.Unlock()    return c.count}```### 3.3 RWMutex
读写锁，适合读多写少场景：```go
type Cache struct {    mu    sync.RWMutex    data map[string]string}func (c *Cache) Get(key string) string {    c.mu.RLock()    defer c.mu.RUnlock()    return c.data[key]}func (c *Cache) Set(key, value string) {    c.mu.Lock()    defer c.mu.Unlock()    c.data[key] = value}```### 3.4 Once
只执行一次：```go
var once sync.Oncevar instance *Databasefunc getDatabase() *Database {    once.Do(func() {        instance = &Database{...}    })    return instance}```## 4. Context
### 4.1 Context 用法
用于传递截止时间、取消信号：```go
func main() {    ctx, cancel := context. WithTimeout(context.Background(), 2*time.Second)    defer cancel()        select {    case <-time.After(1 * time.Second):        fmt.Println("finished in 1s")    case <-ctx.Done():        fmt.Println("timeout:", ctx.Err())    }}```### 4.2 传递 Context
```go
func search(query string, ctx context.Context) ([]Result, error) {    // 可以通过 ctx 取消    select {    case <-ctx.Done():        return nil, ctx.Err()    default:        // 执行搜索    }}func main() {    ctx, cancel := context.WithCancel(context.Background())    results, _ := search("golang", ctx)    cancel()  // 取消}```> ✅ **最佳实践**：在 HTTP/gRPC 服务中，context 用于传递请求级别的取消和超时。## 5. 并发模式
### 5.1 Pipeline
```go
func gen(nums ...int) <-chan int {    out := make(chan int)    go func() {        for _, n := range nums {            out <- n        }        close(out)    }()    return out}func sq(in <-chan int) <-chan int {    out := make(chan int)    go func() {        for n := range in {            out <- n * n        }        close(out)    }()    return out}func main() {    for n := range sq(gen(1, 2, 3, 4, 5)) {        fmt.Println(n)    }}```### 5.2 Fan-out / Fan-in
```go
// 多个 worker 并行处理func main() {    in := gen(1, 2, 3, 4, 5)        // Fan-out: 3 个 worker    c1 := square(in)    c2 := square(in)    c3 := square(in)        // Fan-in: 合并结果    for n := range merge(c1, c2, c3) {        fmt.Println(n)    }}```### 5.3 Worker Pool
```gofunc worker(id int, jobs <-chan int, results chan<- int) {    for j := range jobs {        results <- j * 2    }}func main() {    jobs := make(chan int, 100)    results := make(chan int, 100)        // 启动 3 个 worker    for w := 1; w <= 3; w++ {        go worker(w, jobs, results)    }        // 发送 9 个任务    for j := 1; j <= 9; j++ {        jobs <- j    }    close(jobs)        // 收集结果    for r := 1; r <= 9; r++ {        <-results    }}```## 6. 小结
